function bridgeHtml_() {
  return '<!DOCTYPE html><html><head><meta charset="UTF-8"><title>WP</title></head><body><script>' +
    '(function(){' +
    'function send(m){try{parent.postMessage(m,"*");}catch(e){}}' +
    'window.addEventListener("message",function(e){' +
      'var d=e.data;if(!d||d.t!=="wp0")return;' +
      'try{' +
        'google.script.run' +
          '.withSuccessHandler(function(r){send({t:"wp1",id:d.id,ok:true,res:r});})' +
          '.withFailureHandler(function(err){send({t:"wp1",id:d.id,ok:false,err:String(err&&err.message||err)});})' +
          '.apiFromClient(d.a,d.p||{});' +
      '}catch(err){send({t:"wp1",id:d.id,ok:false,err:String(err&&err.message||err)});}' +
    '});' +
    'send({t:"wp-ready"});' +
    '})();' +
    '</script></body></html>';
}

function doGet(e) {
  var p = parseParams_(e);
  if (String(p.bridge || '') === '1') {
    return HtmlService.createHtmlOutput(bridgeHtml_())
      .setTitle('WarungPOS')
      .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL);
  }
  if (p.action || p.a || p.callback) {
    return respond_(handle_(p), e);
  }
  var tpl = HtmlService.createTemplateFromFile('Index');
  tpl.GAS_URL = ScriptApp.getService().getUrl();
  return tpl.evaluate()
    .setTitle('WarungPOS')
    .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL)
    .addMetaTag('viewport', 'width=device-width, initial-scale=1, viewport-fit=cover');
}

function apiFromClient(action, payload) {
  if (action && typeof action === 'object' && payload == null) {
    payload = action.payload || action.data || {};
    action = action.action || action.a || '';
  }
  var r = handle_({ action: action, payload: payload || {} });
  try { return JSON.parse(JSON.stringify(r)); } catch (e) { return r; }
}

function doPost(e) {
  let params = {};
  try {
    if (e.postData && e.postData.contents) {
      const raw = e.postData.contents;
      try {
        params = JSON.parse(raw);
      } catch (err) {
        params = parseQuery_(raw);
      }
    }
  } catch (err) {
    params = {};
  }
  Object.assign(params, parseParams_(e));
  return respond_(handle_(params), e);
}

function parseParams_(e) {
  const p = {};
  if (e && e.parameter) {
    Object.keys(e.parameter).forEach(function (k) { p[k] = e.parameter[k]; });
  }
  return p;
}

function parseQuery_(s) {
  const o = {};
  String(s || '').split('&').forEach(function (pair) {
    const i = pair.indexOf('=');
    if (i > -1) o[decodeURIComponent(pair.slice(0, i))] = decodeURIComponent(pair.slice(i + 1).replace(/\+/g, ' '));
  });
  return o;
}

function respond_(data, e) {
  const cb = e && e.parameter && e.parameter.callback;
  const json = JSON.stringify(data);
  if (cb) {
    return ContentService.createTextOutput(cb + '(' + json + ')').setMimeType(ContentService.MimeType.JAVASCRIPT);
  }
  return ContentService.createTextOutput(json).setMimeType(ContentService.MimeType.JSON);
}

function ok_(data) { return { ok: true, data: data }; }
function fail_(msg) { return { ok: false, error: String(msg || 'Terjadi kesalahan') }; }

function handle_(p) {
  try {
    ensureSetup_();
    const action = String(p.action || p.a || '').trim();
    let payload = p.payload || p.data || p;
    if (typeof payload === 'string') {
      try { payload = JSON.parse(payload); } catch (e) { /* keep */ }
    }
    switch (action) {
      case 'ping': return ok_({ pong: true, waktu: nowIso_() });
      case 'setup': return ok_(ensureSetup_(true));
      case 'bootstrap': return ok_(bootstrap_());
      case 'login': return login_(payload);
      case 'dashboard': return ok_(dashboard_());
      case 'getPengaturan': return ok_(getPengaturan_());
      case 'savePengaturan': return savePengaturan_(payload);
      case 'getUsers': return ok_(rows_('User'));
      case 'saveUser': return saveUser_(payload);
      case 'deleteUser': return deleteById_('User', payload.id);
      case 'getKategori': return ok_(rows_('Kategori'));
      case 'saveKategori': return saveRow_('Kategori', payload, ['nama', 'warna', 'aktif']);
      case 'deleteKategori': return deleteById_('Kategori', payload.id);
      case 'getJenis': return ok_(rows_('Jenis'));
      case 'saveJenis': return saveRow_('Jenis', payload, ['nama', 'kode', 'deskripsi', 'aktif']);
      case 'deleteJenis': return deleteById_('Jenis', payload.id);
      case 'getSupplier': return ok_(rows_('Supplier'));
      case 'saveSupplier': return saveRow_('Supplier', payload, ['nama', 'telp', 'alamat', 'aktif']);
      case 'deleteSupplier': return deleteById_('Supplier', payload.id);
      case 'getPelanggan': return ok_(rows_('Pelanggan'));
      case 'savePelanggan': return savePelanggan_(payload);
      case 'deletePelanggan': return deleteById_('Pelanggan', payload.id);
      case 'getProduk': return ok_(rows_('Produk'));
      case 'saveProduk': return saveProduk_(payload);
      case 'deleteProduk': return deleteById_('Produk', payload.id);
      case 'adjustStok': return adjustStok_(payload);
      case 'penjualan': return penjualan_(payload);
      case 'getPenjualan': return ok_(listTrans_('Penjualan', payload));
      case 'getPenjualanDetail': return ok_(detailTrans_('Penjualan', 'PenjualanItem', 'penjualan_id', payload.id));
      case 'pembelian': return pembelian_(payload);
      case 'getPembelian': return ok_(listTrans_('Pembelian', payload));
      case 'getPembelianDetail': return ok_(detailTrans_('Pembelian', 'PembelianItem', 'pembelian_id', payload.id));
      case 'returPenjualan': return returPenjualan_(payload);
      case 'getReturPenjualan': return ok_(listTrans_('ReturPenjualan', payload));
      case 'getReturPenjualanDetail': return ok_(detailTrans_('ReturPenjualan', 'ReturPenjualanItem', 'retur_id', payload.id));
      case 'returPembelian': return returPembelian_(payload);
      case 'getReturPembelian': return ok_(listTrans_('ReturPembelian', payload));
      case 'getReturPembelianDetail': return ok_(detailTrans_('ReturPembelian', 'ReturPembelianItem', 'retur_id', payload.id));
      case 'getHutang': return ok_(ringkasanHutang_());
      case 'getHutangDetail': return ok_(detailHutang_(payload.pelanggan_id || payload.id));
      case 'bayarHutang': return bayarHutang_(payload);
      case 'getLaporan': return ok_(laporan_(payload));
      case 'getLedger': return ok_(ledger_());
      case 'getKas': return ok_(rows_('Kas'));
      case 'saveKas': return saveKas_(payload);
      case 'getTutupKasir': return ok_(rows_('TutupKasir'));
      case 'saveTutupKasir': return saveTutupKasir_(payload);
      case 'opname': return opname_(payload);
      case 'getOpname': return ok_(listTrans_('Opname', payload || {}));
      case 'getOpnameDetail': return ok_(detailTrans_('Opname', 'OpnameItem', 'opname_id', payload.id));
      default: return fail_('Aksi tidak dikenal: ' + action);
    }
  } catch (err) {
    return fail_(err.message || err);
  }
}

/* ===================== SETUP ===================== */

function ensureSetup_(force) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  Object.keys(HEADERS).forEach(function (name) {
    let sh = ss.getSheetByName(name);
    if (!sh) sh = ss.insertSheet(name);
    const headers = HEADERS[name];
    const existing = sh.getRange(1, 1, 1, Math.max(headers.length, sh.getLastColumn() || 1)).getValues()[0];
    const empty = !existing[0];
    if (empty || force) {
      sh.getRange(1, 1, 1, headers.length).setValues([headers]);
      sh.getRange(1, 1, 1, headers.length).setFontWeight('bold').setBackground('#0f766e').setFontColor('#ffffff');
      sh.setFrozenRows(1);
    }
  });
  const setSheet = ss.getSheetByName('Pengaturan');
  if (setSheet.getLastRow() < 2) seed_(ss);
  hideGrid_();
  return { spreadsheetId: ss.getId(), nama: ss.getName() };
}

function hideGrid_() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const keep = Object.keys(HEADERS);
  ss.getSheets().forEach(function (sh) {
    if (keep.indexOf(sh.getName()) === -1 && ss.getSheets().length > 1) {
      try { ss.deleteSheet(sh); } catch (e) {}
    }
  });
}

function saveKas_(p) {
  p = p || {};
  var jenis = String(p.jenis || '');
  if (jenis !== 'masuk' && jenis !== 'keluar') return fail_('Jenis kas tidak valid');
  var jumlah = num_(p.jumlah);
  if (jumlah <= 0) return fail_('Nominal tidak valid');
  var row = {
    id: uid_('KS'),
    tanggal: p.tanggal || nowIso_(),
    jenis: jenis,
    jumlah: jumlah,
    keterangan: p.keterangan || '',
    user_id: p.user_id || '',
    referensi: p.referensi || ''
  };
  appendObj_('Kas', row);
  return ok_(row);
}

function tglStr_(v) {
  if (Object.prototype.toString.call(v) === '[object Date]' && !isNaN(v)) {
    return Utilities.formatDate(v, TZ, 'yyyy-MM-dd');
  }
  return String(v || '').slice(0, 10);
}

function saveTutupKasir_(p) {
  p = p || {};
  var tgl = tglStr_(p.tanggal || todayStr_());
  var rincian = p.rincian;
  if (rincian && typeof rincian !== 'string') {
    try { rincian = JSON.stringify(rincian); } catch (e) { rincian = '{}'; }
  }
  var row = {
    id: p.id || uid_('TK'),
    tanggal: tgl,
    modal_awal: num_(p.modal_awal),
    omset: num_(p.omset),
    pengeluaran: num_(p.pengeluaran),
    modal_besok: num_(p.modal_besok),
    setoran: num_(p.setoran),
    rincian: rincian || '{}',
    user_id: p.user_id || '',
    created_at: p.created_at || nowIso_(),
    catatan: p.catatan || ''
  };
  var all = rows_('TutupKasir');
  var exist = null;
  var uidk = String(row.user_id || '');
  for (var i = 0; i < all.length; i++) {
    if (tglStr_(all[i].tanggal) === tgl && String(all[i].user_id || '') === uidk) { exist = all[i]; break; }
  }
  if (exist) {
    row.id = exist.id;
    updateFields_('TutupKasir', exist.id, row);
  } else {
    appendObj_('TutupKasir', row);
  }
  return ok_(row);
}

function opname_(p) {
  const lock = LockService.getScriptLock();
  lock.waitLock(20000);
  try {
    p = p || {};
    const raw = p.items || [];
    const items = [];
    for (let i = 0; i < raw.length; i++) {
      const it = raw[i];
      if (it == null) continue;
      if (it.stok_fisik === '' || it.stok_fisik == null) continue;
      if (String(it.stok_fisik).trim() === '') continue;
      items.push(it);
    }
    if (!items.length) return fail_('Isi stok fisik minimal satu produk');
    const id = uid_('OP');
    const nomor = nomor_('OP');
    const tanggal = p.tanggal || nowIso_();
    const norm = [];
    for (let i = 0; i < items.length; i++) {
      const it = items[i];
      const pr = findById_('Produk', it.produk_id);
      if (!pr) return fail_('Produk tidak ditemukan');
      const fisik = num_(it.stok_fisik);
      if (fisik < 0) return fail_('Stok fisik tidak boleh minus');
      const sist = num_(pr.stok);
      norm.push({
        pr: pr,
        stok_sistem: sist,
        stok_fisik: fisik,
        selisih: roundQty_(fisik - sist)
      });
    }
    appendObj_('Opname', {
      id: id,
      nomor: nomor,
      tanggal: tanggal,
      user_id: p.user_id || '',
      catatan: p.catatan || '',
      jumlah_item: norm.length
    });
    norm.forEach(function (n) {
      appendObj_('OpnameItem', {
        id: uid_('OI'),
        opname_id: id,
        produk_id: n.pr.id,
        nama_produk: n.pr.nama,
        stok_sistem: n.stok_sistem,
        stok_fisik: n.stok_fisik,
        selisih: n.selisih,
        satuan: n.pr.satuan || ''
      });
      updateFields_('Produk', n.pr.id, {
        stok: roundQty_(n.stok_fisik),
        updated_at: nowIso_()
      });
    });
    log_(p.user_id, 'opname', nomor);
    return ok_({ id: id, nomor: nomor, jumlah_item: norm.length });
  } finally {
    lock.releaseLock();
  }
}

function login_(p) {
  const username = String((p && p.username) || '').trim().toLowerCase();
  const password = String((p && (p.password || p.pwd || p.pass)) || '');
  if (!username || !password) return fail_('Username dan password wajib diisi');
  const users = rows_('User');
  const u = users.filter(function (x) {
    return String(x.username).toLowerCase() === username && Number(x.aktif) === 1;
  })[0];
  if (!u) return fail_('User tidak ditemukan atau nonaktif');
  if (String(u.password) !== hash_(password) && String(u.password) !== password) return fail_('Password salah');
  log_((p && p.user_id) || u.id, 'login', u.username);
  return ok_({
    id: u.id,
    username: u.username,
    nama: u.nama,
    role: u.role,
    token: hash_(u.id + '|' + nowIso_())
  });
}

function hash_(s) {
  const raw = Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, String(s), Utilities.Charset.UTF_8);
  return raw.map(function (b) {
    const v = (b < 0 ? b + 256 : b).toString(16);
    return v.length === 1 ? '0' + v : v;
  }).join('');
}

function saveUser_(p) {
  if (!p.username || !p.nama) return fail_('Username dan nama wajib');
  const all = rows_('User');
  const dup = all.filter(function (u) {
    return String(u.username).toLowerCase() === String(p.username).toLowerCase() && u.id !== p.id;
  })[0];
  if (dup) return fail_('Username sudah dipakai');
  if (p.password) p.password = hash_(p.password);
  else if (p.id) {
    const old = all.filter(function (u) { return u.id === p.id; })[0];
    if (old) p.password = old.password;
  }
  if (!p.password) return fail_('Password wajib untuk user baru');
  p.role = p.role || 'kasir';
  p.aktif = p.aktif === 0 || p.aktif === '0' ? 0 : 1;
  p.created_at = p.created_at || nowIso_();
  return saveRow_('User', p, ['username', 'password', 'nama', 'role', 'aktif', 'created_at']);
}

/* ===================== PENJUALAN ===================== */

function penjualan_(p) {
  const lock = LockService.getScriptLock();
  lock.waitLock(20000);
  try {
    const items = p.items || [];
    if (!items.length) return fail_('Keranjang kosong');
    const metode = String(p.metode || 'tunai').toLowerCase();
    const pelangganId = p.pelanggan_id || 'P0';
    if (metode === 'hutang' && (!pelangganId || pelangganId === 'P0')) {
      return fail_('Pelanggan wajib dipilih untuk transaksi hutang');
    }

    const jenisMap = indexBy_(rows_('Jenis'), 'id');
    const produkAll = rows_('Produk');
    const produkMap = indexBy_(produkAll, 'id');

    let subtotal = 0;
    const norm = [];
    for (let i = 0; i < items.length; i++) {
      const it = items[i];
      const pr = produkMap[it.produk_id];
      if (!pr || Number(pr.aktif) === 0) return fail_('Produk tidak valid: ' + (it.nama_produk || it.produk_id));
      const jenis = jenisMap[pr.jenis_id] || {};
      const kode = String(jenis.kode || '').toLowerCase();
      const isDingin = !!it.is_dingin && kode === 'minuman';
      let harga = isDingin ? num_(pr.harga_dingin || pr.harga_jual) : num_(pr.harga_jual);
      let qty = num_(it.qty);
      let modeBerat = '';
      let nominal = 0;

      if (kode === 'berat') {
        modeBerat = String(it.mode_berat || 'berat');
        if (modeBerat === 'nominal') {
          nominal = num_(it.nominal);
          if (nominal <= 0) return fail_('Nominal ' + pr.nama + ' tidak valid');
          if (harga <= 0) return fail_('Harga ' + pr.nama + ' belum diisi');
          qty = roundQty_(nominal / harga);
        } else {
          qty = convertToBase_(num_(it.qty), it.satuan_input || pr.satuan, pr.satuan);
          nominal = 0;
        }
      }
      if (qty <= 0) return fail_('Qty ' + pr.nama + ' tidak valid');
      const line = roundMoney_(kode === 'berat' && modeBerat === 'nominal' ? nominal : qty * harga);
      if (num_(pr.stok) + 1e-9 < qty) return fail_('Stok ' + pr.nama + ' tidak cukup (sisa ' + pr.stok + ' ' + pr.satuan + ')');
      subtotal += line;
      norm.push({
        produk: pr,
        qty: qty,
        satuan: pr.satuan,
        harga_satuan: harga,
        is_dingin: isDingin ? 1 : 0,
        mode_berat: modeBerat,
        nominal: nominal,
        subtotal: line,
        nama_produk: pr.nama
      });
    }

    const diskon = num_(p.diskon);
    const total = roundMoney_(Math.max(0, subtotal - diskon));
    let bayar = num_(p.bayar);
    let kembalian = 0;
    if (metode === 'hutang') {
      bayar = 0;
      kembalian = 0;
    } else if (metode === 'campuran') {
      if (bayar < 0) return fail_('Nominal bayar tidak valid');
      if (bayar > total) return fail_('Bayar campuran tidak boleh lebih dari total');
      kembalian = 0;
    } else if (metode === 'qris' || metode === 'transfer' || metode === 'debit') {
      bayar = total;
      kembalian = 0;
    } else {
      if (bayar < total) return fail_('Uang bayar kurang');
      kembalian = roundMoney_(bayar - total);
    }

    const id = uid_('PJ');
    const nomor = nomor_('PJ');
    const tanggal = p.tanggal || nowIso_();
    appendObj_('Penjualan', {
      id: id, nomor: nomor, tanggal: tanggal, pelanggan_id: pelangganId,
      user_id: p.user_id || '', subtotal: subtotal, diskon: diskon, total: total,
      bayar: bayar, kembalian: kembalian, metode: metode, status: 'selesai',
      catatan: p.catatan || ''
    });

    norm.forEach(function (n) {
      appendObj_('PenjualanItem', {
        id: uid_('PI'), penjualan_id: id, produk_id: n.produk.id, nama_produk: n.nama_produk,
        qty: n.qty, satuan: n.satuan, harga_satuan: n.harga_satuan, is_dingin: n.is_dingin,
        mode_berat: n.mode_berat, nominal: n.nominal, subtotal: n.subtotal
      });
      const sisa = roundQty_(num_(n.produk.stok) - n.qty);
      updateFields_('Produk', n.produk.id, { stok: sisa, updated_at: nowIso_() });
      n.produk.stok = sisa;
    });

    const hutangNominal = metode === 'hutang' ? total : (metode === 'campuran' ? roundMoney_(total - bayar) : 0);
    if (hutangNominal > 0) {
      catatHutang_(pelangganId, 'hutang', id, hutangNominal, 'Belanja ' + nomor, p.user_id);
    }

    log_(p.user_id, 'penjualan', nomor + ' ' + total);
    return ok_({
      id: id, nomor: nomor, tanggal: tanggal, subtotal: subtotal, diskon: diskon,
      total: total, bayar: bayar, kembalian: kembalian, metode: metode,
      pelanggan_id: pelangganId, items: norm.map(function (n) {
        return {
          produk_id: n.produk.id, nama_produk: n.nama_produk, qty: n.qty, satuan: n.satuan,
          harga_satuan: n.harga_satuan, is_dingin: n.is_dingin, mode_berat: n.mode_berat,
          nominal: n.nominal, subtotal: n.subtotal
        };
      })
    });
  } finally {
    lock.releaseLock();
  }
}

function pembelian_(p) {
  const lock = LockService.getScriptLock();
  lock.waitLock(20000);
  try {
    const items = p.items || [];
    if (!items.length) return fail_('Item pembelian kosong');
    if (!p.supplier_id) return fail_('Supplier wajib dipilih');
    let total = 0;
    const norm = [];
    for (let i = 0; i < items.length; i++) {
      const it = items[i];
      const pr = findById_('Produk', it.produk_id);
      if (!pr) return fail_('Produk tidak ditemukan');
      const qty = num_(it.qty);
      const hb = num_(it.harga_beli || pr.harga_beli);
      if (qty <= 0) return fail_('Qty tidak valid');
      const sub = roundMoney_(qty * hb);
      total += sub;
      norm.push({ pr: pr, qty: qty, harga_beli: hb, subtotal: sub });
    }
    const id = uid_('PB');
    const nomor = nomor_('PB');
    appendObj_('Pembelian', {
      id: id, nomor: nomor, tanggal: p.tanggal || nowIso_(), supplier_id: p.supplier_id,
      user_id: p.user_id || '', total: roundMoney_(total), status: 'selesai', catatan: p.catatan || ''
    });
    norm.forEach(function (n) {
      appendObj_('PembelianItem', {
        id: uid_('BI'), pembelian_id: id, produk_id: n.pr.id, nama_produk: n.pr.nama,
        qty: n.qty, harga_beli: n.harga_beli, subtotal: n.subtotal
      });
      updateFields_('Produk', n.pr.id, {
        stok: roundQty_(num_(n.pr.stok) + n.qty),
        harga_beli: n.harga_beli,
        updated_at: nowIso_()
      });
    });
    log_(p.user_id, 'pembelian', nomor);
    return ok_({ id: id, nomor: nomor, total: roundMoney_(total) });
  } finally {
    lock.releaseLock();
  }
}

function returPenjualan_(p) {
  const lock = LockService.getScriptLock();
  lock.waitLock(20000);
  try {
    const pj = findById_('Penjualan', p.penjualan_id);
    if (!pj) return fail_('Transaksi penjualan tidak ditemukan');
    const items = p.items || [];
    if (!items.length) return fail_('Item retur kosong');
    const asal = rows_('PenjualanItem').filter(function (x) { return x.penjualan_id === pj.id; });
    const sudah = rows_('ReturPenjualan').filter(function (r) { return r.penjualan_id === pj.id; });
    const sudahIds = sudah.map(function (r) { return r.id; });
    const sudahItem = rows_('ReturPenjualanItem').filter(function (i) { return sudahIds.indexOf(i.retur_id) !== -1; });
    const returQty = {};
    sudahItem.forEach(function (i) { returQty[i.produk_id] = num_(returQty[i.produk_id]) + num_(i.qty); });

    let total = 0;
    const norm = [];
    for (let i = 0; i < items.length; i++) {
      const it = items[i];
      const line = asal.filter(function (a) { return a.produk_id === it.produk_id; })[0];
      if (!line) return fail_('Produk tidak ada di transaksi asal');
      const qty = num_(it.qty);
      const sisa = num_(line.qty) - num_(returQty[it.produk_id]);
      if (qty <= 0 || qty > sisa + 1e-9) return fail_('Qty retur ' + line.nama_produk + ' melebihi sisa');
      const sub = roundMoney_(qty * num_(line.harga_satuan));
      total += sub;
      norm.push({ line: line, qty: qty, subtotal: sub });
    }
    total = roundMoney_(total);
    const id = uid_('RJ');
    const nomor = nomor_('RJ');
    const metode = String(p.metode_refund || 'tunai');
    appendObj_('ReturPenjualan', {
      id: id, nomor: nomor, penjualan_id: pj.id, tanggal: p.tanggal || nowIso_(),
      user_id: p.user_id || '', total: total, alasan: p.alasan || '', metode_refund: metode
    });
    norm.forEach(function (n) {
      appendObj_('ReturPenjualanItem', {
        id: uid_('RI'), retur_id: id, produk_id: n.line.produk_id, nama_produk: n.line.nama_produk,
        qty: n.qty, harga: n.line.harga_satuan, subtotal: n.subtotal
      });
      const pr = findById_('Produk', n.line.produk_id);
      if (pr) updateFields_('Produk', pr.id, { stok: roundQty_(num_(pr.stok) + n.qty), updated_at: nowIso_() });
    });
    if (metode === 'potong_hutang' && pj.pelanggan_id && pj.pelanggan_id !== 'P0') {
      catatHutang_(pj.pelanggan_id, 'cicilan', id, total, 'Potong hutang retur ' + nomor, p.user_id);
    }
    log_(p.user_id, 'retur_jual', nomor);
    return ok_({ id: id, nomor: nomor, total: total });
  } finally {
    lock.releaseLock();
  }
}

function returPembelian_(p) {
  const lock = LockService.getScriptLock();
  lock.waitLock(20000);
  try {
    const pb = findById_('Pembelian', p.pembelian_id);
    if (!pb) return fail_('Pembelian tidak ditemukan');
    const items = p.items || [];
    if (!items.length) return fail_('Item retur kosong');
    const asal = rows_('PembelianItem').filter(function (x) { return x.pembelian_id === pb.id; });
    let total = 0;
    const norm = [];
    for (let i = 0; i < items.length; i++) {
      const it = items[i];
      const line = asal.filter(function (a) { return a.produk_id === it.produk_id; })[0];
      if (!line) return fail_('Produk tidak ada di pembelian asal');

      const pr = findById_('Produk', it.produk_id);

      const sub = roundMoney_(qty * num_(line.harga_beli));
      total += sub;
      norm.push({ line: line, pr: pr, qty: qty, subtotal: sub });
    }
    const id = uid_('RB');

    appendObj_('ReturPembelian', {

    });
    norm.forEach(function (n) {
      appendObj_('ReturPembelianItem', {
        id: uid_('RBI'), retur_id: id, produk_id: n.line.produk_id, nama_produk: n.line.nama_produk,
        qty: n.qty, harga: n.line.harga_beli, subtotal: n.subtotal

    return ok_({ id: id, nomor: nomor, total: roundMoney_(total) });

    lock.releaseLock();
  }
}

/* ===================== HUTANG ===================== */

function catatHutang_(pelangganId, jenis, referensi, jumlah, keterangan, userId) {
  const pl = findById_('Pelanggan', pelangganId);
  if (!pl) throw new Error('Pelanggan tidak ditemukan');
  const sebelum = num_(pl.saldo_hutang);
  const delta = jenis === 'hutang' ? num_(jumlah) : -num_(jumlah);
  let sesudah = roundMoney_(sebelum + delta);
  if (sesudah < 0) sesudah = 0;
  appendObj_('Hutang', {
    id: uid_('HT'), pelanggan_id: pelangganId, tanggal: nowIso_(), jenis: jenis,
    referensi: referensi || '', jumlah: num_(jumlah), saldo_sebelum: sebelum,
    saldo_sesudah: sesudah, keterangan: keterangan || '', user_id: userId || ''
  });
  updateFields_('Pelanggan', pelangganId, { saldo_hutang: sesudah });
  return sesudah;
}

/* ===================== LAPORAN ===================== */

function listTrans_(sheetName, p) {
  let list = rows_(sheetName);
  const from = p && p.from ? String(p.from) : '';
  const to = p && p.to ? String(p.to) : '';
  if (from) list = list.filter(function (r) { return String(r.tanggal).slice(0, 10) >= from; });
  if (to) list = list.filter(function (r) { return String(r.tanggal).slice(0, 10) <= to; });
  list.sort(function (a, b) { return String(b.tanggal).localeCompare(String(a.tanggal)); });
  return list;
}

function detailTrans_(head, item, fk, id) {
  const h = findById_(head, id);
  if (!h) throw new Error('Data tidak ditemukan');
  const items = rows_(item).filter(function (x) { return x[fk] === id; });
  return { header: h, items: items };
}

function laporan_(p) {
  const jenis = String((p && p.jenis) || 'penjualan');
  const from = (p && p.from) || '';
  const to = (p && p.to) || '';
  const map = {
    penjualan: 'Penjualan',
    pembelian: 'Pembelian',
    retur_penjualan: 'ReturPenjualan',
    retur_pembelian: 'ReturPembelian'
  };
  const sheet = map[jenis];
  if (!sheet) throw new Error('Jenis laporan tidak dikenal');
  const list = listTrans_(sheet, { from: from, to: to });
  const total = list.reduce(function (s, r) { return s + num_(r.total); }, 0);
  return { jenis: jenis, from: from, to: to, rows: list, total: roundMoney_(total), count: list.length };
}

function dashboard_() {
  const today = todayStr_();
  const pj = rows_('Penjualan').filter(function (r) { return String(r.tanggal).slice(0, 10) === today && r.status !== 'batal'; });
  const omzet = pj.reduce(function (s, r) { return s + num_(r.total); }, 0);
  const tunai = pj.filter(function (r) { return r.metode === 'tunai'; }).reduce(function (s, r) { return s + num_(r.total); }, 0);
  const hutangHari = pj.filter(function (r) { return r.metode === 'hutang' || r.metode === 'campuran'; })
    .reduce(function (s, r) {
      if (r.metode === 'hutang') return s + num_(r.total);
      return s + (num_(r.total) - num_(r.bayar));
    }, 0);
  const produk = rows_('Produk').filter(function (p) { return Number(p.aktif) === 1; });
  const menipis = produk.filter(function (p) { return num_(p.stok) <= num_(p.stok_min); });
  const totalHutang = rows_('Pelanggan').reduce(function (s, p) { return s + num_(p.saldo_hutang); }, 0);
  return {
    tanggal: today,
    transaksi_hari_ini: pj.length,
    omzet_hari_ini: roundMoney_(omzet),
    tunai_hari_ini: roundMoney_(tunai),
    hutang_hari_ini: roundMoney_(hutangHari),
    total_produk: produk.length,
    stok_menipis: menipis.length,
    total_piutang: roundMoney_(totalHutang),
    menipis: menipis
  };
}

/* ===================== SHEET HELPERS ===================== */

function sheet_(name) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sh = ss.getSheetByName(name);
  if (!sh) throw new Error('Sheet tidak ada: ' + name);
  return sh;
}

function rows_(name) {
  const sh = sheet_(name);
  const last = sh.getLastRow();
  const lastCol = sh.getLastColumn();
  if (last < 2) return [];
  const values = sh.getRange(1, 1, last, lastCol).getValues();
  const headers = values[0].map(function (h) { return String(h).trim(); });
  const out = [];
  for (let i = 1; i < values.length; i++) {
    const row = values[i];
    if (!row[0]) continue;
    const o = {};
    headers.forEach(function (h, idx) { o[h] = row[idx]; });
    out.push(o);
  }
  return out;
}

function findById_(name, id) {
  if (!id) return null;
  const list = rows_(name);
  for (let i = 0; i < list.length; i++) if (String(list[i].id) === String(id)) return list[i];
  return null;
}

function indexBy_(arr, key) {
  const o = {};
  arr.forEach(function (x) { o[x[key]] = x; });
  return o;
}

function appendObj_(name, obj) {
  const sh = sheet_(name);
  const headers = HEADERS[name];
  const row = headers.map(function (h) { return obj[h] !== undefined && obj[h] !== null ? obj[h] : ''; });
  sh.appendRow(row);
}

function saveRow_(name, p, fields) {
  const headers = HEADERS[name];
  if (!p.id) {
    p.id = uid_(name.slice(0, 2).toUpperCase());
    if (headers.indexOf('aktif') !== -1 && p.aktif === undefined) p.aktif = 1;
    appendObj_(name, p);
    return ok_(p);
  }
  const sh = sheet_(name);
  const data = sh.getDataRange().getValues();
  const hdr = data[0].map(function (h) { return String(h).trim(); });
  const idCol = hdr.indexOf('id');
  for (let i = 1; i < data.length; i++) {
    if (String(data[i][idCol]) === String(p.id)) {
      fields.forEach(function (f) {
        const c = hdr.indexOf(f);
        if (c !== -1 && p[f] !== undefined) sh.getRange(i + 1, c + 1).setValue(p[f]);
      });
      return ok_(p);
    }
  }
  appendObj_(name, p);
  return ok_(p);
}

function updateFields_(name, id, fields) {
  const sh = sheet_(name);
  const data = sh.getDataRange().getValues();
  const hdr = data[0].map(function (h) { return String(h).trim(); });
  const idCol = hdr.indexOf('id');
  for (let i = 1; i < data.length; i++) {
    if (String(data[i][idCol]) === String(id)) {
      Object.keys(fields).forEach(function (f) {
        const c = hdr.indexOf(f);
        if (c !== -1) sh.getRange(i + 1, c + 1).setValue(fields[f]);
      });
      return;
    }
  }
}

function deleteById_(name, id) {
  if (!id) return fail_('ID kosong');
  if (name === 'User' && String(id) === 'U1') return fail_('User admin utama tidak boleh dihapus');
  if (name === 'Pelanggan' && String(id) === 'P0') return fail_('Pelanggan Umum tidak boleh dihapus');
  const sh = sheet_(name);
  const data = sh.getDataRange().getValues();
  const idCol = 0;
  for (let i = data.length - 1; i >= 1; i--) {
    if (String(data[i][idCol]) === String(id)) {
      sh.deleteRow(i + 1);
      return ok_({ id: id, deleted: true });
    }
  }
  return fail_('Data tidak ditemukan');
}

function putMap_(name, map) {
  const sh = sheet_(name);
  const last = Math.max(sh.getLastRow(), 1);
  if (last > 1) sh.getRange(2, 1, last - 1, 2).clearContent();
  const keys = Object.keys(map);
  if (!keys.length) return;
  const rows = keys.map(function (k) { return [k, map[k]]; });
  sh.getRange(2, 1, rows.length, 2).setValues(rows);
}

function log_(userId, aksi, detail) {
  try {
    appendObj_('Log', { id: uid_('LG'), waktu: nowIso_(), user_id: userId || '', aksi: aksi, detail: String(detail || '') });
  } catch (e) {}
}

/* ===================== UTILS ===================== */

function nowIso_() {
  return Utilities.formatDate(new Date(), TZ, "yyyy-MM-dd'T'HH:mm:ss");
}
function todayStr_() {
  return Utilities.formatDate(new Date(), TZ, 'yyyy-MM-dd');
}
function num_(v) {
  if (v === '' || v === null || v === undefined) return 0;
  const n = Number(String(v).replace(',', '.'));
  return isNaN(n) ? 0 : n;
}
function roundMoney_(n) { return Math.round((Number(n) + Number.EPSILON) * 100) / 100; }
function roundQty_(n) { return Math.round((Number(n) + Number.EPSILON) * 10000) / 10000; }
function uid_(prefix) {
  return prefix + Utilities.formatDate(new Date(), TZ, 'yyyyMMddHHmmss') + Math.floor(Math.random() * 900 + 100);
}
function nomor_(prefix) {
  const d = Utilities.formatDate(new Date(), TZ, 'yyyyMMdd-HHmmss');
  return prefix + '-' + d;
}
function convertToBase_(qty, fromUnit, baseUnit) {
  const f = String(fromUnit || baseUnit || '').toLowerCase();
  const b = String(baseUnit || '').toLowerCase();
  if (f === b) return qty;
  if (b === 'kg' && (f === 'gram' || f === 'g' || f === 'gr')) return qty / 1000;
  if (b === 'gram' && f === 'kg') return qty * 1000;
  if ((b === 'liter' || b === 'l') && (f === 'ml')) return qty / 1000;
  if (b === 'ml' && (f === 'liter' || f === 'l')) return qty * 1000;
  return qty;
}
